  let isDragging = false;
  let hasMoved = false;

  let dragStartY = 0;
  let initialLeft = 0;
  let initialTop = 0;

  function startDrag(e) {
    if (e.button !== 0) return;
    if (e.target.closest('button, select, input, a, #sc-log')) return;

    const rect = panel.getBoundingClientRect();
    dragStartX = e.clientX;
    dragStartY = e.clientY;
    initialLeft = rect.left;
    initialTop = rect.top;
    isDragging = true;
    hasMoved = false;

    panel.style.left = `${initialLeft}px`;
    panel.style.top = `${initialTop}px`;
    panel.style.right = 'auto';
    panel.style.bottom = 'auto';
    panel.style.transition = 'none';

    window.addEventListener('pointermove', onDragMove, true);
    window.addEventListener('pointerup', onDragEnd, true);
    e.preventDefault();
  }

  function onDragMove(e) {
    if (!isDragging) return;
    const dx = e.clientX - dragStartX;
    const dy = e.clientY - dragStartY;

    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
      hasMoved = true;
    }

    const curWidth = panel.offsetWidth;
    const curHeight = panel.offsetHeight;

    const newLeft = Math.max(4, Math.min(window.innerWidth - curWidth - 4, initialLeft + dx));
    const newTop = Math.max(4, Math.min(window.innerHeight - curHeight - 4, initialTop + dy));

    panel.style.left = `${newLeft}px`;
    panel.style.top = `${newTop}px`;
  }

  function onDragEnd() {
    if (!isDragging) return;
    isDragging = false;
    panel.style.transition = '';
    window.removeEventListener('pointermove', onDragMove, true);
    window.removeEventListener('pointerup', onDragEnd, true);
  }

  panel.addEventListener('pointerdown', startDrag);

  // 5. PLATFORM EXTRACTION LOGIC

  // --- TWITTER / X ---
  function findTweetDataFromDOM(el) {
    if (!el) return null;
    const key = Object.keys(el).find(k => k.startsWith('__reactFiber$') || k.startsWith('__reactProps$'));
    if (!key) return null;

    let curr = el[key];
    let depth = 0;
    while (curr && depth < 50) {
      const p = curr.memoizedProps;
      if (p) {
        if (p.tweet && (p.tweet.legacy || p.tweet.rest_id)) return p.tweet;
        if (p.result && (p.result.legacy || p.result.rest_id)) return p.result;
        if (p.legacy && (p.legacy.full_text !== undefined || p.legacy.id_str)) return p;
      }
      curr = curr.return;
      depth++;
    }
    return null;
  }

  function processTweetLegacy(tweetNode) {
    if (!tweetNode) return false;
    const tweet = tweetNode.tweet || tweetNode;
    const legacy = tweet.legacy || tweet;
    const tweetId = legacy.id_str || tweet.rest_id || tweet.id_str;
    if (!tweetId) return false;

    const userResult = tweet.core?.user_results?.result;
    const userLegacy = userResult?.legacy || userResult?.core;
    const authorHandle = userLegacy?.screen_name
      ? `@${userLegacy.screen_name}`
      : (tweet.user?.screen_name ? `@${tweet.user.screen_name}` : '');
    const authorName = userLegacy?.name || tweet.user?.name || authorHandle;

    const mediaList = legacy.extended_entities?.media || legacy.entities?.media || tweet.extended_entities?.media || [];
    const mediaUrls = [];
    let videoPoster = null;
    let videoUrl = null;

    mediaList.forEach(m => {
      const rawUrl = m.media_url_https || m.media_url;
      let highRes = rawUrl;
      if (rawUrl) {
        const match = rawUrl.match(/^https?:\/\/pbs\.twimg\.com\/media\/([^.]+)\.([a-z]+)/i);
        if (match) {
          highRes = `https://pbs.twimg.com/media/${match[1]}?format=${match[2]}&name=orig`;
        }
      }

      if (m.type === 'photo') {
        if (highRes && !mediaUrls.includes(highRes)) mediaUrls.push(highRes);
      } else if (m.type === 'video' || m.type === 'animated_gif') {
        videoPoster = highRes || rawUrl;
        if (m.video_info && Array.isArray(m.video_info.variants)) {
          const mp4s = m.video_info.variants
            .filter(v => v.content_type === 'video/mp4' && v.url)
            .sort((a, b) => (b.bitrate || 0) - (a.bitrate || 0));

          if (mp4s.length > 0) {
            videoUrl = mp4s[0].url;
          } else if (m.video_info.variants[0]?.url) {
            videoUrl = m.video_info.variants[0].url;
          }
        }

        if (videoUrl && !mediaUrls.includes(videoUrl)) {
          mediaUrls.push(videoUrl);
        } else if (videoPoster && !mediaUrls.includes(videoPoster)) {
          mediaUrls.push(videoPoster);
        }
      }
    });

    const existing = scrapedMap.get(tweetId);
    if (existing && existing.videoUrl && existing.mediaUrls && existing.mediaUrls.length >= mediaList.length) return true;

    scrapedMap.set(tweetId, {
      authorHandle: authorHandle,
      authorName: authorName,
      id: tweetId,
      isArticle: false,
      isVideo: !!videoUrl,
      linkCard: null,
      mediaUrl: videoUrl || mediaUrls[0] || null,
      mediaUrls: mediaUrls,
      quotedTweet: null,
      scrapedAt: new Date().toISOString(),
      text: legacy.full_text || tweet.text || '',
      timestamp: legacy.created_at ? new Date(legacy.created_at).toISOString() : new Date().toISOString(),
      url: authorHandle ? `https://x.com/${authorHandle.replace('@', '')}/status/${tweetId}` : `https://x.com/i/status/${tweetId}`,
      videoPoster: videoPoster,
      videoUrl: videoUrl
    });

    saveToDisk();
    updateCounters();
    return true;
  }

  function extractTweetElement(article) {

  function toProxyMasterUrl(url) {
    if (!url) return '';
    return url
      .replace('i.pximg.net', 'i.pixiv.re')
      .replace(/\/c\/\d+x\d+[^/]*\//, '/')
      .replace(/_(?:square|custom)1200\./, '_master1200.');
  }

  function generatePixivMediaUrls(primaryUrl, pageCount = 1) {
    const highRes = toProxyMasterUrl(primaryUrl);
    if (!highRes) return [];
    if (pageCount <= 1 || !highRes.includes('_p0_')) return [highRes];

    const urls = [];
    for (let i = 0; i < pageCount; i++) {
      urls.push(highRes.replace('_p0_', `_p${i}_`));
    }
    return urls;
  }

  // --- DEVIANTART ---
  function enhanceWixmpUrl(rawUrl) {
    if (!rawUrl || !rawUrl.startsWith('http')) return { url: null, fallback_url: null };

    const tokenMatch = rawUrl.match(/token=([a-zA-Z0-9_\-\.]+)/);
    if (!tokenMatch) return { url: rawUrl, fallback_url: rawUrl };

    try {
      const payloadB64 = tokenMatch[1].split('.')[1].replace(/-/g, '+').replace(/_/g, '/');
      const payload = JSON.parse(atob(payloadB64));
      const obj = payload?.obj?.[0]?.[0];

      if (obj && obj.path) {
        if (payload.aud && payload.aud.includes("urn:service:file.download")) {
          const directUrl = `https://images-wixmp-ed30a86b8c4ca887773594c2.wixmp.com${obj.path}?token=${tokenMatch[1]}`;
          return { url: directUrl, fallback_url: directUrl };
        }
        const maxW = obj.width ? parseInt(obj.width.replace(/[^\d]/g, ''), 10) : 1280;
        const maxH = obj.height ? parseInt(obj.height.replace(/[^\d]/g, ''), 10) : 1280;
        const highRes = rawUrl.replace(/\/v1\/(?:fill|crop|fit)\/[^\/]+(?=\/)/i, `/v1/fit/w_${maxW},h_${maxH},q_95`);
        return { url: highRes, fallback_url: rawUrl };
      }
    } catch (e) {}

    return { url: rawUrl, fallback_url: rawUrl };
  }

  function extractDeviantArtPage(html, doc, fallbackThumb, pageUrl) {
    let title = "Untitled";
    let authorName = "DeviantArt Artist";
    let publishedTime = new Date().toISOString();
    const mediaUrls = [];

    const titleEl = doc.querySelector('h1') || doc.querySelector('meta[property="og:title"]');
    if (titleEl) title = titleEl.textContent || titleEl.getAttribute("content") || title;

    const urlMatch = pageUrl.match(/deviantart\.com\/([^\/]+)\/art\//i);
    if (urlMatch) {
      authorName = urlMatch[1];
    } else {
      const aLink = doc.querySelector('a[data-username]') || doc.querySelector('a.user-link');
      if (aLink) authorName = aLink.getAttribute("data-username") || aLink.textContent.trim();
    }

    const timeEl = doc.querySelector('time[datetime]');
    if (timeEl) publishedTime = timeEl.getAttribute('datetime');

    let p1 = null;
    const contentMatch = html.match(/"contentUrl":\s*"(https:\/\/images-wixmp-[^"]+)"/);
    if (contentMatch) p1 = contentMatch[1].replace(/\\u002F/g, "/");

    if (!p1) {
      const ogMatch = html.match(/<meta\s+property="og:image"\s+content="([^"]+)"/i);
      if (ogMatch) p1 = ogMatch[1].replace(/&amp;/g, "&");
    }

    if (!p1 && fallbackThumb) p1 = fallbackThumb;
    if (p1) {
      const enhanced = enhanceWixmpUrl(p1);
      mediaUrls.push(enhanced.url || p1);
    }

    const allHeaders = Array.from(doc.querySelectorAll("h2"));
    const allImagesH2 = allHeaders.find(h => h.textContent.trim().toLowerCase() === "all images");
    const carouselSection = allImagesH2 ? allImagesH2.closest("section") : null;

    let thumbs = [];
    if (carouselSection) {
      thumbs = Array.from(carouselSection.querySelectorAll("button img, .nijSZp img"))
        .map(img => img.getAttribute("src"))
        .filter(src => src && src.startsWith("http"));
    }

    if (thumbs.length === 0) {
      const preloads = Array.from(doc.querySelectorAll('link[rel="preload"][as="image"]'));
      thumbs = preloads
        .map(l => l.getAttribute("href"))
        .filter(href => href && href.startsWith("http") && !href.includes(".woff") && !href.includes("avatar"));
    }

    if (thumbs.length > 1) {
      for (let i = 1; i < thumbs.length; i++) {
        const enh = enhanceWixmpUrl(thumbs[i]);
        if (enh.url) mediaUrls.push(enh.url);
      }
    }

    return { title, authorName, authorHandle: `@${authorName}`, publishedTime, mediaUrls };
  }

      for (const item of uncollected) {

            url: item.url,
            videoPoster: null

      if (!match) return log("❌ Not on DeviantArt post URL.");
      const id = match[1];
      const ext = extractDeviantArtPage(document.documentElement.outerHTML, document, null, loc);
      const postObj = {
        authorHandle: ext.authorHandle,
        authorName: ext.authorName,
        id: 'da_' + id,
        isArticle: false,
        linkCard: null,
        mediaUrl: ext.mediaUrls[0] || null,
        mediaUrls: ext.mediaUrls,
        fallbackUrl: ext.mediaUrls[0] || null,
        quotedTweet: null,
        scrapedAt: new Date().toISOString(),
        text: ext.title,
        timestamp: ext.publishedTime,
        url: loc,
        videoPoster: null
      };
      scrapedMap.set(id, postObj);
      selectedMap.set('da_' + id, postObj);
      saveToDisk();
      updateCounters();
      log(`[✓] Captured DA: "${ext.title}"`);
    } else if (currentMode === 'twitter') {
      const match = loc.match(/\/status\/(\d+)/);
      if (!match) return log("❌ Not on Tweet status URL.");
      const id = match[1];
      const article = document.querySelector(`article[data-testid="tweet"] a[href*="${id}"]`)?.closest('article') || document.querySelector('article[data-testid="tweet"]');
      if (article && extractTweetElement(article)) {
        selectedMap.set(id, scrapedMap.get(id));
        saveToDisk();
        updateCounters();
        log(`[✓] Captured tweet (#${id}).`);
      } else {
        log("❌ Unable to parse tweet DOM.");
      }
    } else if (currentMode === 'pixiv') {
      const match = loc.match(/\/artworks\/(\d+)/);
      if (!match) return log("❌ Not on artwork URL.");
      const id = match[1];
      const meta = document.getElementById('meta-preload-data');
      if (meta && meta.content) {
        try {
          const data = JSON.parse(meta.content);
          if (data?.illust?.[id]) {
            processPixivWork(data.illust[id]);
            selectedMap.set(id, scrapedMap.get(id));
            saveToDisk();
            updateCounters();
            return log(`[✓] Captured Pixiv post #${id}.`);
          }
        } catch (e) {}
      }
      harvestPixivDOM();
      if (scrapedMap.has(id)) {
        selectedMap.set(id, scrapedMap.get(id));
        saveToDisk();
        updateCounters();
      }
      log(`[✓] Processed Pixiv artwork #${id}.`);
    } else if (currentMode === 'pinterest') {
      const match = loc.match(/\/pin\/(\d+)/);
      if (!match) return log("❌ Not on Pin URL.");
      harvestPinterestDOM();
      const id = match[1];
      if (scrapedMap.has(id)) {
        selectedMap.set(id, scrapedMap.get(id));
        saveToDisk();
        updateCounters();
      }
      log(`[✓] Processed Pin #${id}.`);
    }
  }

  function getPostTarget(node) {
    if (!node || node === panel || panel.contains(node)) return null;
    if (currentMode === 'twitter') return node.closest('article[data-testid="tweet"]');
    if (currentMode === 'safebooru' || currentMode === 'rule34') return node.closest('span.thumb, article.thumb');
    if (currentMode === 'pixiv') return node.closest('a[href*="/artworks/"]') || node.closest('li');
    if (currentMode === 'pinterest') return node.closest('div[data-test-id="pin"], div[data-test-id="pinWrapper"]');
    if (currentMode === 'deviantart') return node.closest('a[href*="/art/"]') || node.closest('section');
    return null;
  }

  function handlePickerMove(e) {
    const target = getPostTarget(e.target);
    if (hoveredEl && hoveredEl !== target) {
      hoveredEl.removeAttribute('data-sc-hovered');
      hoveredEl = null;
    }
    if (target && target !== hoveredEl) {
      hoveredEl = target;
      hoveredEl.setAttribute('data-sc-hovered', 'true');
    }
  }

  async function handlePickerClick(e) {
    if (panel.contains(e.target)) return;
    e.preventDefault();
    e.stopPropagation();

    const target = getPostTarget(e.target);
    if (!target) return log("No post element under pointer.");

    let key = null;

    if (currentMode === 'twitter') {
      extractTweetElement(target);
      const timeEl = target.querySelector('time');
      const href = timeEl?.closest('a')?.getAttribute('href') || '';
      const match = href.match(/\/status\/(\d+)/);
      if (match) key = match[1];
    } else if (currentMode === 'safebooru') {
      key = extractSafebooruThumb(target);
    } else if (currentMode === 'rule34') {
      key = extractRule34Thumb(target);
    } else if (currentMode === 'pixiv') {
      extractPixivCard(target);
      const link = target.matches('a[href*="/artworks/"]') ? target : target.querySelector('a[href*="/artworks/"]');
      const match = (link?.getAttribute('href') || '').match(/\/artworks\/(\d+)/);
      if (match) key = match[1];
    } else if (currentMode === 'pinterest') {
      extractPinterestCard(target);
      const link = target.querySelector('a[href*="/pin/"]') || target.closest('a[href*="/pin/"]');
      const match = (link?.getAttribute('href') || '').match(/\/pin\/(\d+)/);
      if (match) key = match[1];
    } else if (currentMode === 'deviantart') {
      const link = target.matches('a[href*="/art/"]') ? target : target.querySelector('a[href*="/art/"]');
      if (link && link.href) {
        const match = link.href.match(/\/art\/.*?-(\d+)/);
        const daId = match ? match[1] : null;
        if (daId && !scrapedMap.has('da_' + daId)) {
          try {
            const res = await fetch(link.href, { credentials: "include" });
            const html = await res.text();
            const doc = new DOMParser().parseFromString(html, "text/html");
            const ext = extractDeviantArtPage(html, doc, null, link.href);
            scrapedMap.set('da_' + daId, {
              authorHandle: ext.authorHandle,
              authorName: ext.authorName,
              id: 'da_' + daId,
              isArticle: false,
              linkCard: null,
              mediaUrl: ext.mediaUrls[0] || null,
              mediaUrls: ext.mediaUrls,
              fallbackUrl: ext.mediaUrls[0] || null,
              quotedTweet: null,
              scrapedAt: new Date().toISOString(),
              text: ext.title,
              timestamp: ext.publishedTime,
              url: link.href,
              videoPoster: null
            });
            saveToDisk();
          } catch(err) {}
        }
        if (daId) key = 'da_' + daId;
      }
    }

    if (key && scrapedMap.has(key)) {
      if (selectedMap.has(key)) {
        selectedMap.delete(key);
        target.removeAttribute('data-sc-selected');
        log(`[−] Deselected #${key}`);
      } else {
        selectedMap.set(key, scrapedMap.get(key));
        target.setAttribute('data-sc-selected', 'true');
        log(`[✓] Selected #${key}`);
      }
      saveToDisk();
      updateCounters();
    } else {
      log("❌ Failed extracting element.");
    }
  }

  function handlePickerKey(e) {
    if (e.key === 'Escape') stopPicker();
  }

  function startPicker() {
    if (isPicking) return stopPicker();
    isPicking = true;
    pickBtn.innerText = 'Done (Esc)';
    pickBtn.style.background = '#da3633';
    document.body.style.cursor = 'crosshair';

    document.addEventListener('mousemove', handlePickerMove, true);
    document.addEventListener('click', handlePickerClick, true);
    document.addEventListener('keydown', handlePickerKey, true);
    log("Selection Mode: Click post to toggle select/deselect. Press Esc or 'Done' to exit.");
  }

  function stopPicker() {
    isPicking = false;
    pickBtn.innerText = 'Select Posts';
    pickBtn.style.background = '#d29922';
    document.body.style.cursor = 'default';

    if (hoveredEl) {
      hoveredEl.removeAttribute('data-sc-hovered');
      hoveredEl = null;
    }

    document.removeEventListener('mousemove', handlePickerMove, true);
    document.removeEventListener('click', handlePickerClick, true);
    document.removeEventListener('keydown', handlePickerKey, true);
  }

  // 6. UNIVERSAL TRAFFIC HOOKS
  function recursiveJsonScan(obj) {
    // Only capture background network traffic if auto-scroll is actively running
    if (!isScraping || !obj || typeof obj !== 'object') return;

    if (currentMode === 'twitter') {
      if ((obj.legacy && (obj.legacy.id_str || obj.rest_id || obj.legacy.full_text !== undefined)) || (obj.rest_id && obj.legacy)) {
        processTweetLegacy(obj);
        return;
      }
    } else if (currentMode === 'pixiv') {
      if (obj.illustType !== undefined && obj.id && (obj.url || obj.title)) {
        processPixivWork(obj);
        return;
      }
    } else if (currentMode === 'pinterest') {
      if (obj.id && (obj.images || obj.image_spec || obj.type === 'pin')) {
        processPin(obj);
        return;
      }
    }

    if (Array.isArray(obj)) {
      for (const item of obj) recursiveJsonScan(item);
    } else {
      for (const key of Object.keys(obj)) {
        recursiveJsonScan(obj[key]);
      }
    }
  }

  const originalFetch = window.fetch;
  window.fetch = async function(...args) {
    const res = await originalFetch.apply(this, args);
    try {
      if (isScraping) {
        const url = typeof args[0] === 'string' ? args[0] : (args[0]?.url || '');
        if (
          (currentMode === 'twitter' && (url.includes('/graphql/') || url.includes('/Likes') || url.includes('/UserTweets'))) ||
          (currentMode === 'pixiv' && url.includes('/ajax/')) ||
          (currentMode === 'pinterest' && (url.includes('/resource/') || url.includes('Resource/get')))
        ) {
          res.clone().json().then(data => recursiveJsonScan(data)).catch(() => {});
        }
      }
    } catch (e) {}
    return res;
  };

  const origOpen = XMLHttpRequest.prototype.open;
  const origSend = XMLHttpRequest.prototype.send;
  XMLHttpRequest.prototype.open = function(method, url, ...rest) {
    this._url = url;
    return origOpen.apply(this, [method, url, ...rest]);
  };
  XMLHttpRequest.prototype.send = function(...args) {
    this.addEventListener('load', function() {
      try {
        if (isScraping && this._url && currentMode === 'pinterest' && (this._url.includes('/resource/') || this._url.includes('Resource/get'))) {
          recursiveJsonScan(JSON.parse(this.responseText));
        }
      } catch (e) {}
    });
    return origSend.apply(this, args);
  };

  // 7. MASTER HARVESTER DISPATCHER
  function harvestCurrentMode() {
    if (currentMode === 'twitter') harvestTwitterDOM();
    else if (currentMode === 'safebooru') harvestSafebooruDOM();
    else if (currentMode === 'rule34') harvestRule34DOM();
    else if (currentMode === 'pixiv') harvestPixivDOM();
    else if (currentMode === 'pinterest') harvestPinterestDOM();
    else if (currentMode === 'deviantart') harvestDeviantArt();
  }

  // 8. BACKGROUND WORKER & AUTO-SCROLLER
  const workerBlob = new Blob([`
    let timer = null;
    self.onmessage = (e) => {
      if (e.data === 'start') {
        timer = setInterval(() => postMessage('tick'), 600);
      } else if (e.data === 'stop') {
        clearInterval(timer);
      }
    };
  `], { type: 'application/javascript' });
  const worker = new Worker(URL.createObjectURL(workerBlob));

  function tick() {
    harvestCurrentMode();

    const preScroll = window.scrollY;
    window.scrollBy({ top: 1400, behavior: 'instant' });

    if (currentMode === 'rule34' || currentMode === 'safebooru') {
      const isBottom = (window.innerHeight + window.scrollY) >= (document.body.offsetHeight - 400);
      if (isBottom) {
        const nextLink = document.querySelector('a.paginator-next, a[rel="next"], a[alt="next"], a[name="next"], #paginator a:last-child');
        if (nextLink && nextLink.href) {
          log("Next page navigating...");
          nextLink.click();
          return;
        }
      }
    }

    if (window.scrollY === preScroll && !document.hidden) {
      idleCount++;

      if (idleCount > 2) {
        document.querySelectorAll('div[role="button"], span').forEach(b => {
          if (/retry|try again|see new tweets/i.test(b.innerText || '')) {
            try { b.click(); } catch (e) {}
          }
        });

        window.scrollBy(0, -350);
        setTimeout(() => {
          window.scrollBy(0, 500);
          window.dispatchEvent(new Event('scroll'));
        }, 120);
      }

      if (idleCount > 75) {
        stop();
        log("Bottom reached or rate limited.");
      }
    } else {
      idleCount = 0;
    }
  }

  function start() {
    isScraping = true;
    toggleBtn.innerText = 'Pause Auto-Scroll';
    toggleBtn.style.background = '#da3633';
    badgeEl.innerText = 'Running';
    badgeEl.style.background = '#238636';
    harvestCurrentMode();
    worker.postMessage('start');
    log("Auto-scroll started.");
  }

  function stop() {
    isScraping = false;
    toggleBtn.innerText = 'Start Auto-Scroll';
    toggleBtn.style.background = '#238636';
    badgeEl.innerText = 'Paused';
    badgeEl.style.background = '#21262d';
    worker.postMessage('stop');
    log("Auto-scroll paused.");
  }

  function closeProgram() {
    stop();
    stopPicker();
    clearAllDomOutlines();

    window.fetch = originalFetch;
    XMLHttpRequest.prototype.open = origOpen;
    XMLHttpRequest.prototype.send = origSend;

    worker.terminate();
    panel.remove();
    styleTag.remove();
    window.__universalArtScraperActive = false;
    console.log("[SCRAPER] Closed and detached.");
  }

  worker.onmessage = (e) => {
    if (e.data === 'tick' && isScraping) tick();
  };

  toggleBtn.onclick = () => (isScraping ? stop() : start());
  rescanBtn.onclick = () => {
    harvestCurrentMode();
    log("Scanned visible elements.");
  };
  pageBtn.onclick = () => fetchCurrentPage();
  pickBtn.onclick = () => startPicker();
  closeBtn.onclick = closeProgram;

  exportBtn.onclick = () => {
    if (scrapedMap.size === 0) return alert("No items recorded yet.");
    const data = { posts: Array.from(scrapedMap.values()) };
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${currentMode}_all_export_${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
    log(`Exported all ${scrapedMap.size} posts.`);
  };

  exportSelectedBtn.onclick = () => {
    if (selectedMap.size === 0) return alert("No posts selected.");
    const data = { posts: Array.from(selectedMap.values()) };
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${currentMode}_selected_export_${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
    log(`Exported ${selectedMap.size} selected posts.`);
  };

  clearBtn.onclick = () => {
    if (confirm(`Clear cached items for ${currentMode.toUpperCase()}?`)) {
      clearAllDomOutlines();
      localStorage.removeItem(STORAGE_PREFIX + currentMode);
      localStorage.removeItem(SELECTED_PREFIX + currentMode);
      scrapedMap.clear();
      selectedMap.clear();
      updateCounters();
      log("Mode cache cleared.");
    }
  };

  clearSelectedBtn.onclick = () => {
    clearAllDomOutlines();
    selectedMap.clear();
    saveToDisk();
    updateCounters();
    log("Selected items cleared.");
  };

  // Only read existing cache into UI on boot; do not scrape
  updateCounters();
  log(`Ready on ${window.location.hostname}`);
})();
